给你两棵二叉树的根节点 `p` 和 `q` ，编写一个函数来检验这两棵树是否相同。

如果两个树在结构上相同，并且节点具有相同的值，则认为它们是相同的。


```ts
/**
 * Definition for a binary tree node.
 * class TreeNode {
 *     val: number
 *     left: TreeNode | null
 *     right: TreeNode | null
 *     constructor(val?: number, left?: TreeNode | null, right?: TreeNode | null) {
 *         this.val = (val===undefined ? 0 : val)
 *         this.left = (left===undefined ? null : left)
 *         this.right = (right===undefined ? null : right)
 *     }
 * }
 */

function isSameTree(p: TreeNode | null, q: TreeNode | null): boolean {
    //一次遍历，可以递归的看两个数是否是相同的树
    //终止条件：q.val != p.val ，直接 return false
    //正常是相等的话，比较左右子树是否相等，需要左右子树都要相等才能判定当前字数相等
    //都为 null 的话直接返回 true
    if(q === null && p === null) return true
    if((q == null && p !=null) || (p==  null && q!= null) ) return false
    if(p && q &&  q.val != p.val) return false 
    
    const left_res = isSameTree(p.left,q.left)
    const right_res = isSameTree(p.right,q.right)

    return left_res && right_res
};
```
